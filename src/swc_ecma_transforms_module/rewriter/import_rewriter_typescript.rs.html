14struct Rewriter;

16impl VisitMut for Rewriter {
17    fn visit_mut_call_expr(&mut self, e: &mut CallExpr) {
18        e.visit_mut_children_with(self);
19
20        // In TypeScript, `require` is handled only in JavaScript files.
21        // I don't know how to check if the current file is a JavaScript file in swc,
22        // So I'll temporarily just simplify.
23        if !e.callee.is_import()
24            && !e
25                .callee
26                .as_expr()
27                .and_then(|e| e.as_ident())
28                .is_some_and(|i| i.sym == "require")
29        {
30            return;
31        }
32
33        if let Some(ExprOrSpread { spread: None, expr }) = &mut e.args.get_mut(0) {
34            if let Expr::Lit(Lit::Str(s)) = &mut **expr {
35                if let Some(src) = get_output_extension_wtf8(&s.value) {
36                    s.raw = None;
37                    s.value = src;
38                }
39            } else {
40                let importee = expr.take();
41                *expr = Box::new(Expr::Call(CallExpr {
42                    span: DUMMY_SP,
43                    callee: helper!(ts_rewrite_relative_import_extension),
44                    args: vec![ExprOrSpread {
45                        spread: None,
46                        expr: importee,
47                    }],
48                    ..Default::default()
49                }));
50            }
51        }
52    }
53
54    fn visit_mut_import_decl(&mut self, i: &mut ImportDecl) {
55        if let Some(src) = get_output_extension_wtf8(&i.src.value) {
56            i.src.value = src;
57            i.src.raw = None;
58        }
59    }
60
61    fn visit_mut_named_export(&mut self, e: &mut NamedExport) {
62        if let Some(src) = &mut e.src {
63            if let Some(new_src) = get_output_extension_wtf8(&src.value) {
64                src.value = new_src;
65                src.raw = None;
66            }
67        }
68    }
69
70    fn visit_mut_export_all(&mut self, n: &mut ExportAll) {
71        if let Some(new_src) = get_output_extension_wtf8(&n.src.value) {
72            n.src.value = new_src;
73            n.src.raw = None;
74        }
75    }
76}

78fn get_output_extension(specifier: &Atom) -> Option<Atom> {
79    // https://github.com/microsoft/TypeScript/blob/3eb7b6a1794a6d2cde7948a3016c57e628b104b9/src/compiler/utilities.ts#L4331
80    let path = Path::new(specifier.as_str());
81    if !(path.starts_with("./")
82        || path.starts_with("../")
83        || path.starts_with(".\\")
84        || path.starts_with("..\\"))
85    {
86        return None;
87    }
88
89    // https://github.com/microsoft/TypeScript/blob/3eb7b6a1794a6d2cde7948a3016c57e628b104b9/src/compiler/parser.ts#L10556
90    if path.file_name()?.to_str()?.contains(".d.") {
91        return None;
92    }
93
94    // https://github.com/microsoft/TypeScript/blob/3eb7b6a1794a6d2cde7948a3016c57e628b104b9/src/compiler/emitter.ts#L540
95    let ext = path.extension()?.to_str()?;
96    let ext = match ext {
97        "json" => "json",
98        "jsx" | "tsx" => "jsx",
99        "mjs" | "mts" => "mjs",
100        "cjs" | "cts" => "cjs",
101        _ => "js",
102    };
103
104    Some(Atom::new(path.with_extension(ext).to_str()?))
105}

107fn get_output_extension_wtf8(specifier: &Wtf8Atom) -> Option<Wtf8Atom> {
108    let normalized = specifier.to_atom_lossy();
109    get_output_extension(normalized.as_ref()).map(Into::into)
